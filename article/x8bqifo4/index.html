		<script>
			/**
			 * js 弱类型语言 不需要定义数据类型，在执行时自动识别
			 * js 解释性语言 脚本语言
			 * 计算机不能直接理解高级语言，只能直接理解机器语言，所以必须要把高级语言翻译成机器语言，计算机才能执行高级语言编写的程序。
			 * 一个是编译，一个是解释。两种方式只是翻译的时间不同。
			 * 编译型语言写的程序执行之前，需要一个专门的编译过程，把程序编译成为机器语言的文件，比如exe文件，以后要运行的话就不用重新翻译了，直接使用编译的结果就行了（exe文件），
			 * 因为翻译只做了一次，运行时不需要翻译，所以编译型语言的程序执行效率高,
			 * 但也不能一概而论，部分解释型语言的解释器通过在运行时动态优化代码，甚至能够使解释型语言的性能超过编译型语言。
			 * 解释则不同，解释性语言的程序不需要编译，省了道工序，解释性语言在运行程序的时候才翻译，比如解释性basic语言，专门有一个解释器能够直接执行basic程序，每个语句都是执行的时候才翻译。
			 * 这样解释性语言每执行一次就要翻译一次，效率比较低。解释是一句一句的翻译。
			 */
			/**
			 * js 定义变量的关键词 var
			 * es6 不推荐使用var,推荐使用let
			 * var 定义的变量 全局变量  在整个js的范围都可以使用
			 * 如果在定义变量前使用变量的值为undefined 
			 * let 定义的变量 局部变量 必须先定义后使用
			 * const 定义常量
			 */
			let  x = 1;
			const y = 2;
			console.log(x+y);
			console.log(typeof x); // 查看变量的数据类型
			/** js 数据类型
			 * 1 number 数字(整数 小数)
			 * 2 string 字符串
			 * 3 boolean 布尔值
			 * 4 undefined 未定义的变量
			 * 5 function (******)
			 * 6 object 对象类型
			 */
			x = 'abc'; 
			console.log(typeof x);
			/** 
			 * 多个值同时进行定义赋值 解构赋值
			 */ 
			let [a1,a2,a3,a4] = [1,2,3,4];
			console.log(a1,a2,a3,a4); // 同时打印四个值	
			[a1,a2,a3,a4] = "abcd"; // 字符串本来就是数组 
			console.log(a1,a2,a3,a4);
			/**
			 * 对象的结构赋值
			 * js中内置的数据类型 -> Date 日期
			 */
			console.log(typeof Date); // 类型时function(native function) 是系统自带的获取时间的api
			console.log(Date());
			let now = new Date(); // 如果使用new关键字 则now是对象 否则now是函数的返回值，类型为字符串
			console.log(now.getFullYear());
			/** js 方法
			 * 1 定义方法
			 * 	 关键字 方法名(形参){方法逻辑}
			 * 2 调用方法
			 * 	 方法名();
			 * js 不能进行方法重载 使用内置arguments对象进行重载实现
			 */
			function add(){ // ES6使用 rest参数 ...形参名
				console.log(arguments); // js方法内置对象，获取不确定的实参 类型为数组 
				if(arguments.length == 0) console.log("add方法没有传参");
				else {
					let sum = 0;
					for(let i in arguments) sum+=arguments[i];
					console.log(`传入的参数总和为${sum}`);
				}
			}
			add(10,200,300);
			/**
			 * 函数声明 两种方式
			 * 1 function 函数名(){} 优先加载 优先分配
			 * 2 let 函数名 = function(){} 必须先定义才能使用
			 * 3 箭头函数（es6新增) const 函数名 = ()=>{} 同2先定义在使用
			 */
			let test = function(){
				console.log("test函数调用");
			}
			test();
			test1();
			function test1(){
				console.log("test1函数调用");
			}
			const test2 = ()=>{
				console.log("test函数调用");
			}
			test2();
			/**
			 * 通过function进行一类事物的表示
			 */
			function Cat(name,age,type){
				this.name = name;
				this.age = age;
				this.type = type;
				this.run = function(){
					console.log("run方法被执行了");
				}
			}
			let cat = new Cat("test","2","花猫");
			console.log(typeof cat);
			cat.run();
			console.log(cat.name) // 属性访问
			cat.color = "黑色"; // 对象的属性如果不存在，则进行属性自动的添加
			console.log(cat); 
			for(let [key,value] of Object.entries(cat)) console.log(key,value); // 查看一个对象包含了多少属性多少方法
			for(let i in cat)  console.log(i,cat[i]); // for in 循环也可以
			/**
			 * json表达对象,使用花括号进行结构
			 */
			const dog = {
				name:'xiaobai',
				age:'5',
				color:'white'
			};
			
			let {name,age,color} = dog; // 对象解构
			console.log(name,age,color);
			const dog2 = { // 对象的组装
				name:name, // 可以简化为 name,
				age:age, // 可以简化 age,
				color:color ,// 可以简化color
				bark:()=>{
					console.log("dog2正在叫");
				}
			}
			console.log(dog2);
			dog2.bark();
			/**
			 * string常用方法
			 */
			let str = "abcdaadcdaaabcd";
			console.log(str.charAt(3)); // 获取指定位置的字符
			console.log(str.at(3)); // 获取指定位置的字符
			console.log(str.startsWith("a")); // 查看是否以指定字符串开头
			console.log(str.endsWith("d")); // 查看是否以指定字符串开头
			console.log(str.toLowerCase()); // 转小写
			console.log(str.toUpperCase()); // 转大写
			console.log(str.substring(1,4)); // 字符串截取左闭右开
			let s = "2"
			console.log(s.padStart(2,'0')); // 如果字符串长度不够的话，在前面补字符0
			let date = new Date();
			let pat =  `${date.getFullYear()}-${date.getMonth().toString().padStart(2,'0')}-${date.getDay()}`// 字符串模板 es6
			console.log(pat); // 格式化日期字符串
			str.trim(); // 去除前后空格
			console.log(str.replace("a","X")); // 用X替换字符串中所有的a字符
			let reg = new RegExp(/a?/); // 正则表达式
			console.log(reg.test(str)); // 正则表达式是否匹配
			console.log(str.replace(/a+/g,"X")); // 使用正则表达式来替换,必须使用全局正则表达式所以最后要加个g
		</script> <!-- js 通过浏览器的javascript引擎执行，为了防止浏览器交替执行，所以一般js代码放到最后 --> 
	</body>
