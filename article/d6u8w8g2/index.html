    @ApiOperation("查看学生答题成绩")
    @PostMapping("/queryTeachCaseStudentList")
    public Result queryTeachCaseStudentList(@RequestBody QueryTeachCaseStudentListVo queryTeachCaseStudentListVo) {
        QueryWrapper<Account> classQueryWrapper = new QueryWrapper<>();
        QueryWrapper<TeachCaseStudent> idQueryWrapper = new QueryWrapper<>();
        if (queryTeachCaseStudentListVo.getClassCode() != null)
            classQueryWrapper.eq("class_code", queryTeachCaseStudentListVo.getClassCode());
        List<Long> idList = accountService.list(classQueryWrapper).stream().map(Account::getId).collect(Collectors.toList());
        if (!idList.isEmpty()) idQueryWrapper.in("student_id", idList);
        else return ResultUtil.success(ResultCodeEnum.SUCCESS.getCode(), "查询结果为空");
        Page<TeachCaseStudent> resultPage = new Page<>(queryTeachCaseStudentListVo.getPageNumber(), queryTeachCaseStudentListVo.getPageSize());
        teachCaseStudentService.page(resultPage, idQueryWrapper);
        return ResultUtil.success(ResultCodeEnum.SUCCESS, resultPage);
    }

ResultVO

结果封装类

package xyz.hmz.springbootdemo1.vo;

import lombok.AllArgsConstructor;
import lombok.Data;

/**
 * @Author haomingze
 * @Date 2025/3/10 11:23
 * @description: 全局返回结果封装
 */
@Data
@AllArgsConstructor
public class ResultVO {
    /**
     * 业务的状态值
     * 0 表示正常
     * 1 表示有问题
     */
    private  Integer code;
    /**
     * 对状态值的描述
     */
    private  String message;
    /**
     * 携带的数据
     */
    private  Object data;
    public ResultVO(Integer code,String message){
        this.code = code;
        this.message = message;
    }
    private static final ResultVO DEFAULT_SUCCESS = new ResultVO(0,"操作成功");
    private static final ResultVO DEFAULT_FAIL = new ResultVO(1,"操作失败");
    public static ResultVO success(){
        return DEFAULT_SUCCESS;
    }
    public static ResultVO fail(){
        return DEFAULT_FAIL;
    }
    public static ResultVO success(String message){
        return new ResultVO(0,message);
    }
    public static ResultVO fail(String message){
        return new ResultVO(1,message);
    }
    public static ResultVO success(Object data){
        return success("操作成功",data);
    }
    public static ResultVO fail(Object data){
        return fail("操作失败",data);
    }
    public static ResultVO success(String message,Object data){
        return new ResultVO(0,message,data);
    }
    public static ResultVO fail(String message,Object data){
        return new ResultVO(1,message,data);
    }
}

ResultControllerAdvice

返回体拦截，统一返回个人感觉没必要，容易造成返回混乱

package xyz.hmz.springbootdemo1.controller;

import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.SneakyThrows;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.core.MethodParameter;
import org.springframework.http.MediaType;
import org.springframework.http.converter.HttpMessageConverter;
import org.springframework.http.server.ServerHttpRequest;
import org.springframework.http.server.ServerHttpResponse;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.servlet.mvc.method.annotation.ResponseBodyAdvice;
import xyz.hmz.springbootdemo1.vo.ResultVO;

/**
 * @Author haomingze
 * @Date 2025/3/10 15:32
 * @description: 全局结果统一拦截
 */
@RestControllerAdvice
public class ResultControllerAdvice implements ResponseBodyAdvice<Object> {

    @Autowired
    private ObjectMapper objectMapper;

    @Override
    public boolean supports(MethodParameter returnType, Class<? extends HttpMessageConverter<?>> converterType) {
        return true;
    }

    @SneakyThrows
    @Override
    public Object beforeBodyWrite(Object body, MethodParameter returnType, MediaType selectedContentType, Class<? extends HttpMessageConverter<?>> selectedConverterType, ServerHttpRequest request, ServerHttpResponse response) {
        if(body instanceof String) return objectMapper.writeValueAsString(ResultVO.success(body));
        if(body instanceof ResultVO) return body;
        return ResultVO.success(body);
    }
}

ExceptionControllerAdvice

接口层异常捕获

package xyz.hmz.springbootdemo1.controller;

import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import xyz.hmz.springbootdemo1.vo.ResultVO;

/**
 * @Author haomingze
 * @Date 2025/3/3 11:25
 * @description: 全局异常捕获
 */

@RestControllerAdvice
public class ExceptionControllerAdvice {
    @ExceptionHandler(Exception.class)
    public ResultVO handle(){
        return ResultVO.fail();
    }
}

Maven打包设置主类

在pom.xml文件中插件项添加插件

<plugin>
	<groupId>org.apache.maven.plugins</groupId>
    <artifactId>maven-assembly-plugin</artifactId>
    <executions>
    	<execution>
        	<phase>package</phase>
            <goals>
            	<goal>single</goal>
            </goals>
            <configuration>
            	<archive>
                	<manifest>
                    	<mainClass>
                        	xyz.hmz.backend.BackendApplication  # 替换为启动类
                        </mainClass>
                    </manifest>
                </archive>
            	<descriptorRefs>
                	<descriptorRef>jar-with-dependencies</descriptorRef>
                </descriptorRefs>
            </configuration>
        </execution>
    </executions>
</plugin>
