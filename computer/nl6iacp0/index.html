前言

为什么要数组来模拟各个数据结构呢？而不是使用结构体+指针的方式来实现各种数据结构呢？

因为当我们要生成我们的数据结构的时候，需要调用new Node()；方法 这个操作是非常耗时间的，当数据量比较大的时候，如果进行大量的new操作，会非常的耗时，影响效率。

    q[++tt] = x;
}

    hh++;

    }
}

// (1) 拉链法
int h[N], e[N], ne[N], idx;

// 向哈希表中插入一个数
void insert(int x)
{
    int k = (x % N + N) % N;
    e[idx] = x;
    ne[idx] = h[k];
    h[k] = idx ++ ;
}

// 在哈希表中查询某个数是否存在
bool find(int x)
{
    int k = (x % N + N) % N;
    for (int i = h[k]; i != -1; i = ne[i])
        if (e[i] == x)
            return true;

    return false;
}

// (2) 开放寻址法
    int h[N];

// 如果x在哈希表中，返回x的下标；如果x不在哈希表中，返回x应该插入的位置
int find(int x)
{
    int t = (x % N + N) % N;
    while (h[t] != null && h[t] != x)
        {
            t ++ ;
            if (t == N) t = 0;
        }
    return t;
}

int son[N][26], cnt[N], idx;
// 0号点既是根节点，又是空节点
// son[][]存储树中每个节点的子节点
// cnt[]存储以每个节点结尾的单词数量

// 插入一个字符串
void insert(char *str)
{
    int p = 0;
    for (int i = 0; str[i]; i ++ )
    {
        int u = str[i] - 'a';
        if (!son[p][u]) son[p][u] = ++ idx;
        p = son[p][u];
    }
    cnt[p] ++ ;
}

// 查询字符串出现的次数
int query(char *str)
{
    int p = 0;
    for (int i = 0; str[i]; i ++ )
    {
        int u = str[i] - 'a';
        if (!son[p][u]) return 0;
        p = son[p][u];
    }
    return cnt[p];
}

// h[N]存储堆中的值, h[1]是堆顶，x的左儿子是2x, 右儿子是2x + 1
// ph[k]存储第k个插入的点在堆中的位置
// hp[k]存储堆中下标是k的点是第几个插入的
int h[N], ph[N], hp[N], Size;

// 交换两个点，及其映射关系
void heap_swap(int a, int b){
    swap(ph[hp[a]],ph[hp[b]]);
    swap(hp[a], hp[b]);
    swap(h[a], h[b]);
}

void down(int u)
{
    int t = u;
    if (u * 2 <= size && h[u * 2] < h[t]) t = u * 2;
    if (u * 2 + 1 <= size && h[u * 2 + 1] < h[t]) t = u * 2 + 1;
    if (u != t)
    {
        heap_swap(u, t);
        down(t);
    }
}

#include<iostream>
using namespace std;
const int N = 1e5+10;
int heap[N],Size;
void down(int i){
    int t = i;
    if(2*i<=Size&&heap[t]>heap[2*i]) t = 2*i; // 如果左孩子节点存在并且不是最值的话
    if(2*i+1<=Size&&heap[t]>heap[2*i+1]) t = 2*i+1; // 如果右孩子节点存在并且不是最值的话
    if(i != t){
        swap(heap[i],heap[t]); // 交换
        down(t);  // 递归处理子树
    }
}
int main(){
    int n,m;cin>>n>>m;Size = n;
    for(int i = 1;i <= n;i++) cin>>heap[i];
    for(int i = n/2;i > 0;i--) down(i);
    while(m--){
        cout<<heap[1]<<" ";
        swap(heap[1],heap[Size--]);// 每次都和最后一位元素进行交换
        down(1);
    }
    return 0;
}

#include<iostream>
using namespace std;
const int N = 1e6+10;
char s[N],p[N];int ne[N]; // next数组 就表示如果匹配失败， j应该回溯到的值
int main(){
    int n,m;cin>>m>>p+1>>n>>s+1;
    for(int i = 2,j = 0;i<=m;i++){
        while(j && p[i] != p[j+1]) j = ne[j];
        if(p[i] == p[j+1]) j++;
        ne[i] = j;
    }
    for(int i = 1,j = 0;i<=n;i++){
        while(j && s[i] != p[j+1]) j = ne[j];
        if(s[i] == p[j+1]) j++;
        if(j == m){ //匹配成功输出下标
            cout<<i-j<<" ";
            j = ne[j]; // 继续匹配
        }
    }
    return 0;
}
